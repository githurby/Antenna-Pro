cmake_minimum_required(VERSION 3.16)
project(AntennaPro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(AntennaPro WIN32 src/main.cpp src/AntennaModel.cpp src/GLRenderer.cpp src/MatchDiagram.cpp)
target_include_directories(AntennaPro PRIVATE include resources)
if(WIN32)
  enable_language(RC)
  target_sources(AntennaPro PRIVATE resources/app.rc)
endif()
target_compile_definitions(AntennaPro PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(AntennaPro PRIVATE opengl32 gdi32 comctl32 comdlg32 user32)
if(MINGW)
  target_link_options(AntennaPro PRIVATE -municode)
  target_compile_options(AntennaPro PRIVATE -Wall -Wextra)
endif()
